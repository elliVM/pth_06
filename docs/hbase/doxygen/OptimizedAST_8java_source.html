69 public OptimizedAST(final Expression root) {
70 this.root = root;
71 }
72
73 public Expression transformed() {
74 Expression current = root;
75 Expression last;
76 LOGGER.trace("Start AST:\n {}", new PrintAST(root));
77 int i = 0;
78
79 do { // apply until no optimization changes occur
80 last = current;
81 current = new OptimizedExpression(current).transformed();
82 i++;
83 LOGGER.trace("Optimize run <{}> AST:\n {}", i, new PrintAST(current));
84 }
85 while (!current.equals(last));
86 LOGGER.info("Optimized final AST:\n {}", new PrintAST(current));
87 return current;
88 }
